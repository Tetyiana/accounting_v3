# Часті команди

## Перевірка перед пушем
npm run build
npx eslint src 2>&1 | grep -E "no-undef|rules-of-hooks"

## Коміт і пуш
git add -A
git commit -m "feat: продажі — дерево документів, статус-бейдж, контекстне меню замість кнопок"
git push

## Стан
git status
git log --oneline -5

## Скинути кеш PWA (у консолі браузера)
localStorage.removeItem('db_pending_v1'); location.reload();

## Клон з нуля
git clone https://github.com/Tetyiana/accounting_v3.git
cd accounting_v3
npm install
## Робочий цикл проекту 
npm run check:schema    
npm run build
npx eslint src 2>&1 | grep -E "no-undef|rules-of-hooks"
git add -A && git commit -m "..." && git push

## Оновлення податкових форм
npm run sync:forms
